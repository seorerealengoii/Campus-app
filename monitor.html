function doGet() { var docId = "1YdD3b5C93Q_f8C686LQOl1nGLVAyxG3o"; var doc = DocumentApp.openById(docId); var body = doc.getBody(); var tables = body.getTables(); var result = { A9: {}, A12: {} }; tables.forEach(function(table) { var numRows = table.getNumRows(); var currentRoom = null; for (var i = 0; i < numRows; i++) { var row = table.getRow(i); var rowText = row.getText().toUpperCase(); if (rowText.indexOf("SALA 9") !== -1) currentRoom = "A9"; else if (rowText.indexOf("SALA 12") !== -1) currentRoom = "A12"; var numCells = row.getNumCells(); if (currentRoom && numCells >= 5) { var days = ["SEG", "TER", "QUA", "QUI", "SEX"]; // Descobre o desfasamento caso a 1ª coluna seja o nome da sala/horário var offset = numCells > 5 ? 1 : 0; days.forEach(function(day, idx) { var cellIdx = idx + offset; if (cellIdx < numCells) { var cellText = row.getCell(cellIdx).getText().trim(); if (cellText && cellText.length > 1 && cellText.toUpperCase().indexOf("SALA") === -1) { result[currentRoom][day] = cellText; } } }); } } }); return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
